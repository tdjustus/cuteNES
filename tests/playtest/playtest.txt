# Scripted game checks for `rom_harness playtest --list tests/playtest/playtest.txt`.
#
# ROM paths are relative to this file's directory. Blank lines and text after
# '#' are ignored. Each entry is:
#
#   entry <name>              start an entry (name has no spaces)
#   rom <path>                the cartridge to load
#   expect-error <Name>       loading must fail with this LoadError name
#                             (Ok, NotAnInesFile, TruncatedFile, InvalidHeader,
#                              InvalidBankCounts, UnsupportedMapper, Nes2FieldInvalid)
#   run <frames> [buttons]    run frames holding Port One buttons: "none", or
#                             names joined by '+' (A, B, Select, Start, Up,
#                             Down, Left, Right)
#   expect-mem <hexaddr> <hexval>
#                             CPU-visible byte (side-effect free) must equal value
#   expect-changes <n>        at least n frame-to-frame screen changes during
#                             the most recent run
#   font <ascii|a1>           how expect-text decodes tiles: ascii (default) or
#                             a1 (codes 1..26 are letters A..Z, other printable
#                             ASCII as-is)
#   end                       end of entry
#
# Commercial titles are not checked in. Keep local copies under the
# repository's roms-local/ folder (gitignored) and reference them as
# ../../roms-local/<title>.nes.

entry nestest-runs
rom ../roms/other/nestest.nes
run 120 none
expect-changes 1
end

# Mega Man 2 — this local dump (md5 8e4bc5b03ffbd4ef91400e92e50dd294,
# the GoodNES [!] verified "Megaman II (U)") is the 256KB PRG + 8KB
# CHR-RAM MMC1 board: the full 4-bit PRG bank register spans all 16 banks
# (no SUROM outer banking needed below 512K). The title's Start goes
# straight to the password grid (the game's font happens to sit at ASCII
# tile codes, so the grid is greppable). The gameplay entry types the
# documented Normal password (A4 B3 C2 C3 C4 D4 D5 E3 E5 — 4 bosses
# cleared, 3 E-tanks), lands on the stage select, picks a stage and
# verifies the level actually animates — exercising CHR-RAM upload and
# PRG bank switching across the ROM.

entry mega-man-2-boot-reaches-password-screen
rom ../../roms-local/mega-man-2.nes
run 360 none
expect-changes 1
run 10 Start
run 200 none
expect-text PQRSTUVWXYZ[\]^_
expect-changes 1
end
