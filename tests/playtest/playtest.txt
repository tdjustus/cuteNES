# Scripted game checks for `rom_harness playtest --list tests/playtest/playtest.txt`.
#
# ROM paths are relative to this file's directory. Blank lines and text after
# '#' are ignored. Each entry is:
#
#   entry <name>              start an entry (name has no spaces)
#   rom <path>                the cartridge to load
#   expect-error <Name>       loading must fail with this LoadError name
#                             (Ok, NotAnInesFile, TruncatedFile, InvalidHeader,
#                              InvalidBankCounts, UnsupportedMapper, Nes2FieldInvalid)
#   run <frames> [buttons]    run frames holding Port One buttons: "none", or
#                             names joined by '+' (A, B, Select, Start, Up,
#                             Down, Left, Right)
#   expect-mem <hexaddr> <hexval>
#                             CPU-visible byte (side-effect free) must equal value
#   expect-changes <n>        at least n frame-to-frame screen changes during
#                             the most recent run
#   end                       end of entry
#
# Commercial titles are not checked in. Keep local copies under the
# repository's roms-local/ folder (gitignored) and reference them as
# ../../roms-local/<title>.nes.

entry nestest-runs
rom ../roms/other/nestest.nes
run 120 none
expect-changes 1
end

# MMC1 (mapper 1) coverage, T059. All three boot through several
# PRG-banked screens (verified: constant bank flipping between the title,
# menus and gameplay screens). The Legend of Zelda (USA) (Rev 1):
# 128KB PRG + CHR-RAM + battery. Flow: title -> Start -> file select ->
# Start -> REGISTER YOUR NAME, where A types name letters.

entry zelda-boot-reaches-name-registration
rom ../../roms-local/zelda.nes
run 300 none
expect-changes 1
run 10 Start
run 150 none
run 10 Start
run 120 none
run 20 A
run 30 none
run 20 A
run 30 none
expect-text ijjjjjjjjjjjjjjjjjjjjjk
expect-changes 1
end

# Mega Man 2 — NOTE: this local dump (md5 8e4bc5b03ffbd4ef91400e92e50dd294)
# is a 256KB PRG + CHR-RAM MMC1 build, not Capcom's 128K/128K CHR-ROM board;
# its title's Start goes straight to the password grid (the game's font
# happens to sit at ASCII tile codes, so the grid is greppable).

entry mega-man-2-boot-reaches-password-screen

expect-changes 1
run 10 Start
run 200 none
expect-text PQRSTUVWXYZ[\]^_
expect-changes 1
end
